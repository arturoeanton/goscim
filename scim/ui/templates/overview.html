Quick Verification Commands

# 1. Query SCIM 2.0 ServiceProviderConfig (unauthenticated discovery)
curl -s {{.SCIMBaseURL}}/ServiceProviderConfig | jq .

# 2. Search active Users with an RFC 7644 filter
curl -s -H "Authorization: Bearer {{.SampleTokenHint}}" \
  "{{.SCIMBaseURL}}/Users?filter=active%20eq%20true" | jq .

# 3. Run the built-in IdP Conformance Suite (RFC 7643/7644 + Entra + Okta + Keycloak)
goscim conformance -url {{.SCIMBaseURL}} -token "{{.SampleTokenHint}}" -profile all
